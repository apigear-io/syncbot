let selectedLogs = new Set();
let currentLogContent = '';  // Store full content for filtering

async function loadLogs() {
  try {
    const response = await fetch('/api/activation-logs');
    const logs = await response.json();

    const tbody = document.getElementById('logs-tbody');
    selectedLogs.clear();
    updateBatchActions();

    if (!logs || logs.length === 0) {
      tbody.innerHTML = 'No activation logs found.';
      document.getElementById('logs-count').textContent = '0 logs';
      document.getElementById('select-all').checked = false;
      return;
    }

    tbody.innerHTML = logs.map(log => `
      
        
          
        
        ${escapeHtml(log.filename)}
        ${escapeHtml(log.endpoint_name)}
        ${formatBytes(log.size)}
        ${formatLogDate(log.start_time)}
        
          

            
            
              
            
            
          

        
      
    `).join('');

    document.getElementById('logs-count').textContent = `${logs.length} log${logs.length !== 1 ? 's' : ''}`;
    document.getElementById('select-all').checked = false;
  } catch (error) {
    console.error('Failed to load logs:', error);
    document.getElementById('logs-tbody').innerHTML = 'Failed to load logs';
  }
}

async function viewLog(filename) {
  document.getElementById('view-log-filename').textContent = filename;
  document.getElementById('view-log-content').textContent = 'Loading...';
  document.getElementById('view-log-download').href = `/api/activation-logs/${encodeURIComponent(filename)}/download`;
  document.getElementById('log-filter').value = '';
  document.getElementById('log-line-count').textContent = '';
  currentLogContent = '';

  new bootstrap.Modal(document.getElementById('view-log-modal')).show();

  try {
    // Load up to 512KB for viewing
    const response = await fetch(`/api/activation-logs/${encodeURIComponent(filename)}?tail=524288`);
    const data = await response.json();

    if (response.ok) {
      currentLogContent = data.content || '(empty log)';
      document.getElementById('view-log-content').textContent = currentLogContent;
      updateLogLineCount();
    } else {
      document.getElementById('view-log-content').textContent = 'Error: ' + (data.error || 'Failed to load log');
    }
  } catch (error) {
    document.getElementById('view-log-content').textContent = 'Error: Failed to load log';
  }
}

function filterLogContent() {
  const filterText = document.getElementById('log-filter').value.toLowerCase();
  const display = document.getElementById('view-log-content');

  if (!filterText) {
    display.textContent = currentLogContent;
  } else {
    const filteredLines = currentLogContent.split('\n').filter(line =>
      line.toLowerCase().includes(filterText)
    );
    display.textContent = filteredLines.join('\n') || '(no matches)';
  }
  updateLogLineCount();
}

function clearLogFilter() {
  document.getElementById('log-filter').value = '';
  filterLogContent();
}

function updateLogLineCount() {
  const display = document.getElementById('view-log-content');
  const filterText = document.getElementById('log-filter').value;
  const totalLines = currentLogContent.split('\n').length;
  const displayedLines = display.textContent.split('\n').length;

  if (filterText) {
    document.getElementById('log-line-count').textContent = `${displayedLines} of ${totalLines} lines`;
  } else {
    document.getElementById('log-line-count').textContent = `${totalLines} lines`;
  }
}

function formatLogDate(dateStr) {
  if (!dateStr) return '-';
  const date = new Date(dateStr);
  if (isNaN(date.getTime())) return '-';
  return date.toLocaleString();
}

function toggleSelectAll(checkbox) {
  const checkboxes = document.querySelectorAll('.log-checkbox');
  checkboxes.forEach(cb => {
    cb.checked = checkbox.checked;
    if (checkbox.checked) {
      selectedLogs.add(cb.value);
    } else {
      selectedLogs.delete(cb.value);
    }
  });
  updateBatchActions();
}

function toggleSelection(checkbox) {
  if (checkbox.checked) {
    selectedLogs.add(checkbox.value);
  } else {
    selectedLogs.delete(checkbox.value);
  }

  const allCheckboxes = document.querySelectorAll('.log-checkbox');
  const selectAll = document.getElementById('select-all');
  selectAll.checked = selectedLogs.size === allCheckboxes.length && allCheckboxes.length > 0;

  updateBatchActions();
}

function updateBatchActions() {
  const batchActions = document.getElementById('batch-actions');
  const selectedCount = document.getElementById('selected-count');

  if (selectedLogs.size > 0) {
    batchActions.classList.remove('d-none');
    selectedCount.textContent = selectedLogs.size;
  } else {
    batchActions.classList.add('d-none');
  }
}

async function deleteLog(filename) {
  if (!confirm(`Are you sure you want to delete log "${filename}"?`)) {
    return;
  }

  try {
    const response = await fetch(`/api/activation-logs/${encodeURIComponent(filename)}`, {
      method: 'DELETE'
    });

    if (!response.ok) {
      const error = await response.json();
      throw new Error(error.error || 'Failed to delete log');
    }

    showToast('Deleted', 'Log deleted successfully', 'success');
    loadLogs();
  } catch (error) {
    console.error('Failed to delete log:', error);
    showToast('Error', error.message || 'Failed to delete log', 'danger');
  }
}

async function deleteSelectedLogs() {
  const count = selectedLogs.size;
  if (count === 0) return;

  if (!confirm(`Are you sure you want to delete ${count} log${count > 1 ? 's' : ''}?`)) {
    return;
  }

  const filenames = Array.from(selectedLogs);
  let deleted = 0;
  let failed = 0;

  for (const filename of filenames) {
    try {
      const response = await fetch(`/api/activation-logs/${encodeURIComponent(filename)}`, {
        method: 'DELETE'
      });

      if (response.ok) {
        deleted++;
      } else {
        failed++;
      }
    } catch (error) {
      failed++;
    }
  }

  if (failed > 0) {
    showToast('Warning', `Deleted ${deleted} log${deleted !== 1 ? 's' : ''}, ${failed} failed`, 'warning');
  } else {
    showToast('Deleted', `Deleted ${deleted} log${deleted !== 1 ? 's' : ''} successfully`, 'success');
  }

  loadLogs();
}

// Initial load
loadLogs();
