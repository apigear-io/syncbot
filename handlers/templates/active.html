let processState = 'idle';
let lineCount = 0;
let rawOutput = '';  // Store raw output for filtering

// Load initial state on page load
document.addEventListener('DOMContentLoaded', async function() {
  await loadProcessState();
  await loadBufferedOutput();
});

async function loadProcessState() {
  try {
    const response = await fetch('/api/process');
    const info = await response.json();
    updateProcessState(info);
  } catch (error) {
    console.error('Failed to load process state:', error);
  }
}

async function loadBufferedOutput() {
  try {
    const response = await fetch('/api/process/output');
    const data = await response.json();
    if (data.output) {
      appendOutput(data.output);
    }
  } catch (error) {
    console.error('Failed to load buffered output:', error);
  }
}

function updateProcessState(info) {
  processState = info.state;
  const badge = document.getElementById('process-state-badge');
  const killBtn = document.getElementById('kill-btn');
  const exitCodeEl = document.getElementById('process-exit-code');
  const endpointEl = document.getElementById('active-endpoint-name');

  // Update endpoint name if available from process info
  if (info.endpoint_name) {
    endpointEl.textContent = info.endpoint_name;
  }

  // Update state badge
  switch (info.state) {
    case 'running':
      badge.className = 'badge bg-success text-white';
      badge.innerHTML = 'Running';
      killBtn.disabled = false;
      break;
    case 'stopped':
      badge.className = 'badge bg-secondary text-white';
      badge.textContent = 'Stopped';
      killBtn.disabled = true;
      break;
    case 'killed':
      badge.className = 'badge bg-warning text-dark';
      badge.textContent = 'Killed';
      killBtn.disabled = true;
      break;
    case 'failed':
      badge.className = 'badge bg-danger text-white';
      badge.textContent = 'Failed';
      killBtn.disabled = true;
      break;
    default:
      badge.className = 'badge bg-secondary text-white';
      badge.textContent = 'Idle';
      killBtn.disabled = true;
  }

  // Update exit code
  if (info.exit_code !== undefined && info.exit_code !== null) {
    exitCodeEl.textContent = info.exit_code;
    if (info.exit_code === 0) {
      exitCodeEl.className = 'fw-bold text-success';
    } else {
      exitCodeEl.className = 'fw-bold text-danger';
    }
  } else {
    exitCodeEl.textContent = '-';
    exitCodeEl.className = 'fw-bold';
  }
}

function appendOutput(content) {
  const display = document.getElementById('output-display');
  const autoScroll = document.getElementById('auto-scroll').checked;
  const filterText = document.getElementById('log-filter').value.toLowerCase();

  // Store raw output
  rawOutput += content;

  // Apply filter if set
  if (filterText) {
    const filteredLines = rawOutput.split('\n').filter(line =>
      line.toLowerCase().includes(filterText)
    );
    display.textContent = filteredLines.join('\n');
    lineCount = filteredLines.length;
  } else {
    display.textContent = rawOutput;
    lineCount = (rawOutput.match(/\n/g) || []).length;
  }

  updateOutputInfo();

  // Auto-scroll if enabled
  if (autoScroll) {
    display.scrollTop = display.scrollHeight;
  }
}

function updateOutputInfo() {
  const info = document.getElementById('output-info');
  const filterText = document.getElementById('log-filter').value;
  const totalLines = (rawOutput.match(/\n/g) || []).length;

  if (totalLines === 0) {
    info.textContent = 'No output';
  } else if (filterText) {
    info.textContent = `${lineCount} of ${totalLines} lines (filtered)`;
  } else {
    info.textContent = `${totalLines} lines`;
  }
}

async function killProcess() {
  try {
    const response = await fetch('/api/process/kill', { method: 'POST' });
    const info = await response.json();
    updateProcessState(info);
    showToast('Process Killed', 'The process has been terminated', 'warning');
  } catch (error) {
    showToast('Error', 'Failed to kill process: ' + error.message, 'danger');
  }
}

async function clearOutput() {
  try {
    await fetch('/api/process/output', { method: 'DELETE' });
    document.getElementById('output-display').textContent = '';
    rawOutput = '';
    lineCount = 0;
    updateOutputInfo();
    showToast('Cleared', 'Output buffer cleared', 'info');
  } catch (error) {
    showToast('Error', 'Failed to clear output: ' + error.message, 'danger');
  }
}

// Handle process events from SSE (defined in shared_js.html)
function handleProcessStarted(payload) {
  updateProcessState({
    state: 'running',
    endpoint_name: payload.endpoint_name
  });
  // Clear previous output when new process starts
  document.getElementById('output-display').textContent = '';
  rawOutput = '';
  lineCount = 0;
  updateOutputInfo();
}

function handleProcessOutput(payload) {
  appendOutput(payload.content);
}

function handleProcessStopped(payload) {
  updateProcessState({
    state: payload.state,
    exit_code: payload.exit_code,
    endpoint_name: payload.endpoint_name
  });
}

// Filter functions
function filterOutput() {
  const display = document.getElementById('output-display');
  const filterText = document.getElementById('log-filter').value.toLowerCase();

  if (filterText) {
    const filteredLines = rawOutput.split('\n').filter(line =>
      line.toLowerCase().includes(filterText)
    );
    display.textContent = filteredLines.join('\n');
    lineCount = filteredLines.length;
  } else {
    display.textContent = rawOutput;
    lineCount = (rawOutput.match(/\n/g) || []).length;
  }

  updateOutputInfo();
}

function clearFilter() {
  document.getElementById('log-filter').value = '';
  filterOutput();
}

// Download function
function downloadLog() {
  if (!rawOutput) {
    showToast('Info', 'No output to download', 'info');
    return;
  }

  const filterText = document.getElementById('log-filter').value;
  let content = rawOutput;
  let filename = 'activation-log';

  // Download filtered content if filter is active
  if (filterText) {
    const filteredLines = rawOutput.split('\n').filter(line =>
      line.toLowerCase().includes(filterText.toLowerCase())
    );
    content = filteredLines.join('\n');
    filename += '-filtered';
  }

  // Add timestamp to filename
  const timestamp = new Date().toISOString().replace(/[:.]/g, '-').slice(0, 19);
  filename += `-${timestamp}.txt`;

  // Create and trigger download
  const blob = new Blob([content], { type: 'text/plain' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = filename;
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  URL.revokeObjectURL(url);

  showToast('Downloaded', `Log saved as ${filename}`, 'success');
}
