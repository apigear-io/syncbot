// Commands data from server
const commands = [
  {{range $i, $cmd := .Commands}}{{if $i}},{{end}}
  {id: "{{$cmd.ID}}", name: "{{$cmd.Name}}", description: "{{$cmd.Description}}", command: "{{$cmd.Command}}"}
  {{end}}
];

let executeModal = null;
let deleteModal = null;
let pendingDeleteId = null;

// Generate ID from name (e.g., "Restart Service" -> "restart-service")
function generateIdFromName(name) {
  return name
    .toLowerCase()
    .trim()
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/^-+|-+$/g, '');
}

// Open drawer for adding new command
function openAddCommandDrawer() {
  document.getElementById('command-drawer-title').textContent = 'Add Command';
  document.getElementById('cmd-edit-mode').value = '';
  document.getElementById('cmd-id').value = '';
  document.getElementById('cmd-name').value = '';
  document.getElementById('cmd-description').value = '';
  document.getElementById('cmd-command').value = '';

  const drawer = bootstrap.Offcanvas.getOrCreateInstance(document.getElementById('command-drawer'));
  drawer.show();
}

// Open drawer for editing command
function editCommand(id) {
  const cmd = commands.find(c => c.id === id);
  if (!cmd) return;

  document.getElementById('command-drawer-title').textContent = 'Edit Command';
  document.getElementById('cmd-edit-mode').value = cmd.id;
  document.getElementById('cmd-id').value = cmd.id;
  document.getElementById('cmd-name').value = cmd.name;
  document.getElementById('cmd-description').value = cmd.description || '';
  document.getElementById('cmd-command').value = cmd.command;

  const drawer = bootstrap.Offcanvas.getOrCreateInstance(document.getElementById('command-drawer'));
  drawer.show();
}

// Save command (create or update)
async function saveCommand() {
  const editMode = document.getElementById('cmd-edit-mode').value;
  const isEdit = editMode !== '';

  const name = document.getElementById('cmd-name').value.trim();
  const command = {
    id: isEdit ? document.getElementById('cmd-id').value : generateIdFromName(name),
    name: name,
    description: document.getElementById('cmd-description').value.trim(),
    command: document.getElementById('cmd-command').value.trim()
  };

  if (!command.name || !command.command) {
    showToast('Error', 'Name and Command are required', 'danger');
    return;
  }

  if (!command.id) {
    showToast('Error', 'Name must contain at least one alphanumeric character', 'danger');
    return;
  }

  try {
    const url = isEdit ? `/api/commands/${editMode}` : '/api/commands';
    const method = isEdit ? 'PUT' : 'POST';

    const response = await fetch(url, {
      method: method,
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(command)
    });

    if (!response.ok) {
      const data = await response.json();
      showToast('Error', data.error, 'danger');
      return;
    }

    const drawer = bootstrap.Offcanvas.getInstance(document.getElementById('command-drawer'));
    drawer.hide();

    showToast('Success', isEdit ? 'Command updated' : 'Command created', 'success');
    window.location.reload();
  } catch (error) {
    showToast('Error', 'Failed to save command', 'danger');
  }
}

// Delete command
function confirmDelete(id, name) {
  pendingDeleteId = id;
  document.getElementById('delete-cmd-name').textContent = name;

  if (!deleteModal) {
    deleteModal = new bootstrap.Modal(document.getElementById('deleteModal'));
  }
  deleteModal.show();
}

async function deleteCommand() {
  if (!pendingDeleteId) return;

  try {
    const response = await fetch(`/api/commands/${pendingDeleteId}`, { method: 'DELETE' });

    if (!response.ok) {
      const data = await response.json();
      showToast('Error', data.error, 'danger');
      return;
    }

    deleteModal.hide();
    showToast('Success', 'Command deleted', 'success');
    window.location.reload();
  } catch (error) {
    showToast('Error', 'Failed to delete command', 'danger');
  }

  pendingDeleteId = null;
}

// Execute command
async function executeCommand(id, name) {
  document.getElementById('exec-cmd-name').textContent = name;
  document.getElementById('exec-status').innerHTML =
    '
 Executing command...';
  document.getElementById('exec-output').textContent = '';

  if (!executeModal) {
    executeModal = new bootstrap.Modal(document.getElementById('executeModal'));
  }
  executeModal.show();

  try {
    const response = await fetch(`/api/commands/${id}/execute`, { method: 'POST' });
    const result = await response.json();

    if (result.success) {
      document.getElementById('exec-status').innerHTML =
        'Success';
    } else {
      document.getElementById('exec-status').innerHTML =
        `Failed ${escapeHtml(result.error || '')}`;
    }

    document.getElementById('exec-output').textContent = result.output || '(no output)';
  } catch (error) {
    document.getElementById('exec-status').innerHTML = 'Error';
    document.getElementById('exec-output').textContent = 'Failed to execute command: ' + error.message;
  }
}

// Setup delete button handler
document.getElementById('confirm-delete-btn').addEventListener('click', deleteCommand);
